# BJStock ERD

Phase 1 schema in `bjstock`. This diagram matches `0001_initial_business_schema.sql` plus `0002_phase1_schema_hardening.sql` plus `0003_instrument_master_metadata.sql`.

FK default is `ON DELETE RESTRICT`. The only CASCADE is `STOCK_EVALUATION_DETAILS.evaluation_id`.

```mermaid
erDiagram

    INSTRUMENTS ||--o{ FACTOR_VALUES : has
    FACTOR_DEFINITIONS ||--o{ FACTOR_VALUES : defines

    FACTOR_DEFINITIONS ||--o{ STOCK_EVALUATION_DETAILS : contributes_to

    STRATEGIES ||--o{ STRATEGY_VERSIONS : versions
    STRATEGY_VERSIONS ||--o{ STRATEGY_FACTOR_WEIGHTS : uses
    STRATEGY_VERSIONS ||--o{ STRATEGY_RUNS : executed_as

    STRATEGY_RUNS ||--o{ POSITIONS : projects

        BIGINT id PK
        TEXT market
        TEXT symbol
        TEXT name
        TEXT standard_code
        TEXT board
        TEXT instrument_type
        TEXT sector
        TEXT industry
        TEXT currency
        BOOLEAN is_active
        DATE listed_date
        DATE delisted_date
        TIMESTAMPTZ created_at
        TIMESTAMPTZ updated_at
    }

    MARKET_DAILY_BARS {
        BIGINT id PK
        BIGINT instrument_id FK
        DATE trade_date
        NUMERIC open_price
        NUMERIC high_price
        NUMERIC low_price
        NUMERIC close_price
        BIGINT volume
        NUMERIC trading_value
        TEXT source
        TIMESTAMPTZ collected_at
        TIMESTAMPTZ created_at
    }

    FACTOR_DEFINITIONS {
        BIGINT id PK
        TEXT factor_code
        TEXT factor_name
        TEXT category
        TEXT description
        TEXT value_type
        BOOLEAN higher_is_better
        BOOLEAN is_active
        TIMESTAMPTZ created_at
        TIMESTAMPTZ updated_at
    }

    FACTOR_VALUES {
        BIGINT id PK
        BIGINT instrument_id FK
        BIGINT factor_id FK
        DATE evaluation_date
        NUMERIC raw_value
        NUMERIC normalized_score
        TEXT source
        TEXT calculation_version
        TIMESTAMPTZ created_at
    }

    STRATEGIES {
        BIGINT id PK
        TEXT strategy_code
        TEXT strategy_name
        TEXT description
        BOOLEAN is_active
        TIMESTAMPTZ created_at
        TIMESTAMPTZ updated_at
    }

    STRATEGY_VERSIONS {
        BIGINT id PK
        BIGINT strategy_id FK
        INT version_no
        TEXT description
        NUMERIC buy_threshold
        NUMERIC sell_threshold
        DATE valid_from
        DATE valid_to
        TEXT status
        TIMESTAMPTZ created_at
    }

    STRATEGY_FACTOR_WEIGHTS {
        BIGINT id PK
        BIGINT strategy_version_id FK
        BIGINT factor_id FK
        NUMERIC weight
        NUMERIC min_score
        NUMERIC max_score
        BOOLEAN enabled
        TIMESTAMPTZ created_at
    }

    STRATEGY_RUNS {
        BIGINT id PK
        TEXT run_name
        BIGINT strategy_version_id FK
        TEXT run_type
        DATE start_date
        DATE end_date
        NUMERIC initial_cash
        TEXT status
        TIMESTAMPTZ started_at
        TIMESTAMPTZ ended_at
        TIMESTAMPTZ created_at
        TIMESTAMPTZ updated_at
    }

    STOCK_EVALUATIONS {
        BIGINT id PK
        BIGINT strategy_run_id FK
        BIGINT instrument_id FK
        DATE evaluation_date
        NUMERIC quant_score
        NUMERIC ai_score
        NUMERIC final_score
        TEXT quant_decision
        TEXT final_decision
        TIMESTAMPTZ created_at
    }

    STOCK_EVALUATION_DETAILS {
        BIGINT id PK
        BIGINT evaluation_id FK
        BIGINT factor_id FK
        NUMERIC raw_value
        NUMERIC factor_score
        NUMERIC weight
        NUMERIC weighted_score
        TIMESTAMPTZ created_at
    }

    POSITIONS {
        BIGINT id PK
        BIGINT strategy_run_id FK
        BIGINT instrument_id FK
        NUMERIC quantity
        NUMERIC average_price
        NUMERIC realized_profit
        TIMESTAMPTZ updated_at
    }

    ORDERS {
        BIGINT id PK
        TEXT client_order_id
        BIGINT strategy_run_id FK
        BIGINT instrument_id FK
        BIGINT evaluation_id FK
        TEXT side
        TEXT order_type
        NUMERIC requested_price
        NUMERIC quantity
        TEXT status
        TIMESTAMPTZ created_at
        TIMESTAMPTZ executed_at
        TIMESTAMPTZ cancelled_at
    }

    EXECUTIONS {
        BIGINT id PK
        BIGINT order_id FK
        NUMERIC execution_price
        NUMERIC quantity
        NUMERIC commission
        NUMERIC tax
        NUMERIC slippage
        TIMESTAMPTZ executed_at
        TIMESTAMPTZ created_at
    }

    PORTFOLIO_DAILY_SNAPSHOTS {
        BIGINT id PK
        BIGINT strategy_run_id FK
        DATE snapshot_date
        NUMERIC cash
        NUMERIC market_value
        NUMERIC total_asset
        NUMERIC daily_profit
        NUMERIC daily_return
        NUMERIC cumulative_profit
        NUMERIC cumulative_return
        NUMERIC drawdown
        TIMESTAMPTZ created_at
    }

    AI_ADVICE_REQUESTS {
        BIGINT id PK
        BIGINT evaluation_id FK
        TEXT provider
        TEXT model
        TEXT prompt_version
        TEXT request_payload
        TIMESTAMPTZ requested_at
        TIMESTAMPTZ created_at
    }

    AI_ADVICE_RESULTS {
        BIGINT id PK
        BIGINT request_id FK UK
        TEXT recommendation
        NUMERIC confidence
        TEXT summary
        TEXT reasoning_summary
        TEXT risk_notes
        TEXT raw_response
        BOOLEAN used_in_decision
        TIMESTAMPTZ created_at
    }
```
